cmake_minimum_required(VERSION 3.20)
project(JudgeBox LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "JudgeBox requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
if(MINGW)
    add_compile_options(-Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
    add_link_options(-static -static-libgcc -static-libstdc++)
endif()
add_library(judge_process src/process_win32.cpp)
target_include_directories(judge_process PUBLIC src third_party)
add_library(judge_core src/generator.cpp src/checker.cpp src/dataset.cpp src/compiler.cpp src/judge.cpp src/path_compat.cpp
    src/ziputil.cpp src/submitstore.cpp src/problems.cpp src/problemjudge.cpp)
target_link_libraries(judge_core PUBLIC judge_process shell32 ole32 advapi32)
target_include_directories(judge_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/third_party/testlib" "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS web/index.html web/style.css web/app.js web/editor.js src/JudgeBox.ico src/JudgeBox.manifest src/resources.rc)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/editor.js" JBOX_EDITOR_JS)
string(FIND "${JBOX_EDITOR_JS}" ")JBOXED\"" JBOX_EDITOR_COLLISION)
if(NOT JBOX_EDITOR_COLLISION EQUAL -1)
    message(FATAL_ERROR "web/editor.js contains the asset delimiter")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/editor_asset.hpp" "#pragma once
namespace assets {
inline const char* editorJs = R\"JBOXED(${JBOX_EDITOR_JS})JBOXED\";
}
")
file(READ web/index.html JBOX_HTML)
file(READ web/style.css JBOX_CSS)
file(READ web/app.js JBOX_JS)
configure_file(src/assets.hpp.in generated/assets.hpp @ONLY)

# 静态资源表：web/ 根目录的 js/css/html 与 web/vendor/**/*.js 编进可执行文件，
# 前端新增文件不需要再改 C++（规格 §6.10）。editor.js 有 600 KB，已单独嵌入，跳过。
file(GLOB WEB_ASSET_FILES CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_SOURCE_DIR}/web/*.js"
     "${CMAKE_CURRENT_SOURCE_DIR}/web/*.css"
     "${CMAKE_CURRENT_SOURCE_DIR}/web/*.html")
file(GLOB_RECURSE WEB_VENDOR_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/vendor/*.js")
set(JBOX_WEB_ASSETS "// 由 CMake 在 configure 阶段生成。\n#pragma once\n#include <cstddef>\nnamespace webassets {\nstruct WebAsset { const char* path; const char* mime; const char* data; };\ninline const WebAsset kWebAssets[] = {\n")
foreach(asset IN LISTS WEB_ASSET_FILES WEB_VENDOR_FILES)
    get_filename_component(asset_name "${asset}" NAME)
    if(asset_name STREQUAL "editor.js")
        continue()
    endif()
    file(RELATIVE_PATH asset_relative "${CMAKE_CURRENT_SOURCE_DIR}/web" "${asset}")
    string(REPLACE "\\" "/" asset_relative "${asset_relative}")
    get_filename_component(asset_extension "${asset}" EXT)
    if(asset_extension STREQUAL ".js")
        set(asset_mime "text/javascript; charset=utf-8")
    elseif(asset_extension STREQUAL ".css")
        set(asset_mime "text/css; charset=utf-8")
    else()
        set(asset_mime "text/html; charset=utf-8")
    endif()
    file(READ "${asset}" asset_data)
    string(FIND "${asset_data}" ")JBOXWA\"" asset_collision)
    if(NOT asset_collision EQUAL -1)
        message(FATAL_ERROR "${asset} contains the web asset delimiter")
    endif()
    string(APPEND JBOX_WEB_ASSETS "    {\"/${asset_relative}\", \"${asset_mime}\", R\"JBOXWA(${asset_data})JBOXWA\"},\n")
endforeach()
string(APPEND JBOX_WEB_ASSETS "};\ninline const std::size_t kWebAssetCount = sizeof(kWebAssets) / sizeof(kWebAssets[0]);\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/web_assets.hpp" "${JBOX_WEB_ASSETS}")

# checker 在运行时由外部 g++ 编译，安装包不一定带着源码树，所以 testlib.h 编进二进制。
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/third_party/testlib/testlib.h" JBOX_TESTLIB)
string(FIND "${JBOX_TESTLIB}" ")JBOXTX\"" JBOX_TESTLIB_COLLISION)
if(NOT JBOX_TESTLIB_COLLISION EQUAL -1)
    message(FATAL_ERROR "third_party/testlib/testlib.h contains the asset delimiter")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/testlib_asset.hpp" "#pragma once
namespace testlib_asset {
inline const char* header = R\"JBOXTX(${JBOX_TESTLIB})JBOXTX\";
}
")
add_executable(JudgeBox WIN32 src/main.cpp src/server.cpp src/webview_win32.cpp src/resources.rc)
target_include_directories(JudgeBox PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/include")
target_link_libraries(JudgeBox PRIVATE judge_core ws2_32 bcrypt shell32 ole32 oleaut32 user32 gdi32 advapi32 shlwapi)
if(MINGW)
    target_link_options(JudgeBox PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(fixture tests/fixture.cpp)
    add_executable(process_tests tests/process_tests.cpp)
    target_link_libraries(process_tests PRIVATE judge_process)
    add_test(NAME process COMMAND process_tests $<TARGET_FILE:fixture>)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE judge_core)
    add_test(NAME core COMMAND core_tests)
    add_executable(toolchain_paths_tests tests/toolchain_paths_tests.cpp)
    target_link_libraries(toolchain_paths_tests PRIVATE judge_core)
    add_test(NAME toolchain_paths COMMAND toolchain_paths_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME api COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/api_test.py" $<TARGET_FILE:JudgeBox>)
        set_tests_properties(api PROPERTIES TIMEOUT 180)
        add_test(NAME problem COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/problem_api_test.py" $<TARGET_FILE:JudgeBox>)
        set_tests_properties(problem PROPERTIES TIMEOUT 900)
    endif()
endif()
